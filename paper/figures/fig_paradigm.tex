\begin{tikzpicture}[

  box/.style    ={draw=black!50, fill=white, rounded corners=1.6pt, align=center,
                  inner sep=3pt, minimum height=7.6mm, text width=#1},
  box/.default  =2.2cm,
  obox/.style   ={draw=oursemph, fill=finalres, rounded corners=1.6pt, align=center,
                  inner sep=3pt, minimum height=7.6mm, text width=#1, line width=0.7pt},
  obox/.default =2.2cm,
  tok/.style    ={draw=black!40, fill=white, minimum width=3.3mm,
                  minimum height=3.3mm, inner sep=0pt},
  ktok/.style   ={tok, fill=black!20},
  ctok/.style   ={tok, draw=oursemph, fill=finalres, line width=0.6pt},
  ar/.style     ={->, draw=black!55, line width=0.5pt},
  oar/.style    ={->, draw=oursemph, line width=0.6pt},
  tag/.style    ={font=\scriptsize\bfseries, anchor=west},
  led/.style    ={font=\tiny, text=black!55, anchor=west},
  arl/.style    ={font=\tiny, text=black!55, midway, above=0.5pt}
]

% ---- band marking the proposed lane -------------------------------------
\fill[black!4, rounded corners=2pt] (-0.05,-1.42) rectangle (13.25,-3.15);

% ================= lane A: token-based watermark =========================
\node[tag] at (0.05,0.80) {Token-based watermark};

\node[box=2.0cm] (ga) at (1.35,0) {DLM $\theta$\\[-1pt]{\tiny key biases the\\[-1pt]sampled tokens}};

\foreach \i in {0,...,8}{\node[tok] at ({3.30+0.40*\i},0) {};}
\foreach \i in {1,3,6,8}{\node[ktok] at ({3.30+0.40*\i},0) {};}
\node[led, anchor=north] at (4.90,-0.28) {shaded $=$ keyed token subset};

\node[box=2.9cm] (va) at (9.7,0)
  {count keyed tokens\\[-1pt]{\tiny text and key only}};
\node[box=0.85cm] (pa) at (12.6,0) {$p$-value};

\draw[ar] (ga.east) -- (3.08,0);
\draw[ar] (6.76,0) -- (8.14,0) node[arl] {read tokens};
\draw[ar] (va.east) -- (pa.west);

\node[led] at (0.05,-0.90)
  {Evidence lives in the token identities, so the detector reads text and key alone,
   never runs $\theta$, and transfers to anyone holding the key.};

% ================= lane B: Shibboleth =======================================
\node[tag, text=oursemph] at (0.05,-1.63) {Shibboleth (this work)};

\node[obox=2.0cm] (gb) at (1.35,-2.24)
  {DLM $\theta$, steered\\[-1pt]{\tiny key selects which\\[-1pt]answer is written}};

\foreach \i in {0,...,8}{\node[tok] at ({3.30+0.40*\i},-2.24) {};}
\foreach \i in {2,5}{\node[ctok] at ({3.30+0.40*\i},-2.24) {};}
\node[led, anchor=north] at (4.90,-2.52) {outlined $=$ admitted carriers};

\node[obox=2.9cm] (vb) at (9.7,-2.24)
  {re-mask, re-probe, gate\\[-1pt]%
   {\tiny read which way the likelihood moved}};
\node[obox=0.85cm] (pb) at (12.6,-2.24) {$p$-value};

\draw[oar] (gb.east) -- (3.08,-2.24);
\draw[oar] (6.76,-2.24) -- (8.14,-2.24) node[arl, text=oursemph] {rerun $\theta$};
\draw[oar] (vb.east) -- (pb.west);

\node[led] at (0.05,-2.98)
  {Evidence lives in the relation between the text and $\theta$, so the detector reruns that
   same checkpoint at a cost of 144 masked evaluations.};

\end{tikzpicture}
\caption{\textbf{Two places to keep a watermark.} A token-based watermark biases which tokens
are sampled and recovers that bias by reading the text against the key. Shibboleth instead writes
the answer the key requests to a keyed probe of the context, which makes the evidence a property
of the text together with one checkpoint and obliges the detector to rerun it.}
\label{fig:paradigm}
\end{figure}
